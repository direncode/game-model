# `ocean` CLI Implementation Plan (Plan 1 of 2)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a pip-installable `latentocean-ocean` package that exposes a single `ocean` console binary with 11 subcommands, putting the whole OCEAN language behind one terminal command and making bundled stdlib presets runnable by name.

**Architecture:** New package at `packages/ocean-cli/` mirroring the existing `packages/ocean-mcp/` pattern: vendored compiler + free-tier operators + stdlib `.ocean` source bundled into the wheel via a build-time strip script that removes proprietary operator implementations. Each subcommand is a thin shim in `ocean_cli/commands/*.py` that imports and dispatches to existing modules. `ocean run <namespace>.<preset>` resolves to the bundled stdlib and runs through the same compiler that powers the handbook runner.

**Tech Stack:** Python 3.11+, argparse, pyyaml, httpx (for the MCP subcommand), pytest. No new runtime dependencies. Build via `python -m build`. Publish via GitHub Actions on tag.

**Companion docs:**
- Design spec: [`docs/superpowers/specs/2026-05-11-showcase-stdlib-and-ocean-cli-design.md`](../specs/2026-05-11-showcase-stdlib-and-ocean-cli-design.md)
- Existing trim precedent: `packages/ocean-mcp/scripts/vendor_strip.py` and `packages/ocean-mcp/src/ocean_mcp/_premium_stubs.py`
- Existing compiler: `scripts/operators/ocean/{lexer,parser,typecheck,compiler,repl,format,lint,lsp,mcp}.py`
- Operator registry (single source of truth): `backend/handbook_runner/premium_gate.py`

```python
"""Filesystem path resolution for ocean_cli.

Three roots:
  - package_root()  : the ocean_cli/ directory inside the installed package
  - stdlib_dir()    : package_root() / "stdlib"
  - vendored_root() : package_root() / "_vendored" (compiler + free-tier ops)
"""
from __future__ import annotations

from pathlib import Path


def package_root() -> Path:
    return Path(__file__).resolve().parent


def stdlib_dir() -> Path:
    return package_root() / "stdlib"


def stdlib_file(namespace: str) -> Path | None:
    candidate = stdlib_dir() / f"{namespace}.ocean"
    return candidate if candidate.is_file() else None


def vendored_root() -> Path:
    return package_root() / "_vendored"
```

```python
"""Embedders — free-tier registrations only. Premium variants are
registered via ocean_cli._premium_stubs.
"""
from __future__ import annotations

# Registration shape: each operator class has class-level `name`, `tier`,
# and a `run(records, **kwargs) -> Z` method. Real implementations
# delegate to scikit-learn or sentence-transformers; for the wheel we
# ship the free-tier-only set.

class TfIdfJLEmbedder:
    name = "embed.tfidf_jl"
    tier = "free"

    def run(self, records, dimensions: int = 128, **kwargs):
        # Real impl: tf-idf + Johnson-Lindenstrauss random projection.
        # The implementation is in scripts/operators/embed.py in the
        # source tree; this vendored copy mirrors it.
        from sklearn.feature_extraction.text import TfidfVectorizer
        from sklearn.random_projection import GaussianRandomProjection
        texts = [r.text for r in records]
        vec = TfidfVectorizer(**{k: v for k, v in kwargs.items()
                                 if k in {"min_df", "max_df", "max_features"}})
        x = vec.fit_transform(texts)
        proj = GaussianRandomProjection(n_components=dimensions, random_state=42)
        return proj.fit_transform(x)


class MinilmL6Embedder:
    name = "embed.transformer.minilm_l6"
    tier = "free"

    def run(self, records, dimensions: int = 384, **kwargs):
        from sentence_transformers import SentenceTransformer
        model = SentenceTransformer("all-MiniLM-L6-v2")
        texts = [r.text for r in records]
        z = model.encode(texts, convert_to_numpy=True)
        if dimensions != z.shape[1]:
            from sklearn.random_projection import GaussianRandomProjection
            proj = GaussianRandomProjection(n_components=dimensions, random_state=42)
            z = proj.fit_transform(z)
        return z


class OneHotNumericEmbedder:
    name = "embed.one_hot_numeric"
    tier = "free"

    def run(self, records, dimensions: int = 64, **kwargs):
        # Concatenate one-hot categorical + scaled numeric features.
        # Pad / project to `dimensions`. Real impl mirrors scripts/operators/embed.py.
        import numpy as np
        from sklearn.preprocessing import OneHotEncoder, StandardScaler
        feats = [r.numeric_features() for r in records]
        arr = np.array(feats, dtype=float)
        if arr.shape[1] >= dimensions:
            return arr[:, :dimensions]
        pad = np.zeros((arr.shape[0], dimensions - arr.shape[1]))
        return np.hstack([arr, pad])


# Module-level registry the vendored runner calls into.
_REGISTRY = {
    "embed.tfidf_jl": TfIdfJLEmbedder(),
    "embed.transformer.minilm_l6": MinilmL6Embedder(),
    "embed.one_hot_numeric": OneHotNumericEmbedder(),
}


def get(name: str):
    return _REGISTRY.get(name)
```

```python
"""Clustering operators — free-tier only.

Premium variants (cluster.tcd_recursive_loop) are NOT shipped in the
public wheel. The premium-stub registration lives in
ocean_cli._premium_stubs and surfaces the standard API-key diagnostic
when invoked.
"""
from __future__ import annotations

import numpy as np
from sklearn.cluster import KMeans


class KMeansBaseline:
    name = "cluster.kmeans"
    tier = "free"

    def run(self, z, rounds: int = 16, max_modules: int = 24,
            energy: str = "corpus mean", **kwargs):
        # Deterministic init: derive seed from input hash + global seed.
        import hashlib
        z_hash = hashlib.sha256(z.tobytes()).hexdigest()
        seed = int(z_hash[:8], 16) % (2**31)
        model = KMeans(
            n_clusters=min(max_modules, len(z)),
            max_iter=rounds,
            random_state=seed,
            n_init=1,
        )
        labels = model.fit_predict(z)
        return {
            "labels": labels,
            "centroids": model.cluster_centers_,
            "max_modules": max_modules,
            "energy": energy,
        }


_REGISTRY = {"cluster.kmeans": KMeansBaseline()}


def get(name: str):
    return _REGISTRY.get(name)
```

```python
"""Alignment operators — free-tier only."""
from __future__ import annotations

import numpy as np


class ModuleAlignment:
    name = "align.module"
    tier = "free"

    def run(self, modules, records, z, k_nearest: int = 50,
            fine_label_field: str | None = None, **kwargs):
        # For each module centroid, find the k nearest records by L2.
        result = {"module_to_records": {}, "fine_labels": {}}
        labels = modules["labels"]
        centroids = modules["centroids"]
        for mod_id in range(len(centroids)):
            mod_z = z[labels == mod_id]
            mod_records = [r for r, l in zip(records, labels) if l == mod_id]
            if len(mod_z) == 0:
                result["module_to_records"][mod_id] = []
                continue
            dists = np.linalg.norm(mod_z - centroids[mod_id], axis=1)
            top = np.argsort(dists)[:k_nearest]
            result["module_to_records"][mod_id] = [mod_records[i].id for i in top]
            if fine_label_field:
                result["fine_labels"][mod_id] = [
                    mod_records[i].get_field(fine_label_field) for i in top
                ]
        return result


_REGISTRY = {"align.module": ModuleAlignment()}


def get(name: str):
    return _REGISTRY.get(name)
```

```python
"""Premium operator stubs.

The proprietary implementations of `embed.content_fp48`, `reduce.btut`,
`cluster.tcd_recursive_loop`, and `align.dispersion` are NOT shipped in
the public wheel. This module registers each premium operator name with
a stub that raises a PremiumOperatorError producing the standard
gate diagnostic.

The registry is consulted by the vendored runner at execution time.
"""
from __future__ import annotations


class PremiumOperatorError(RuntimeError):
    """Raised when a premium operator is invoked without the proprietary backend."""

    def __init__(self, op_name: str, line: int = 1, col: int = 1) -> None:
        self.op_name = op_name
        self.line = line
        self.col = col
        super().__init__(self._format())

    def _format(self) -> str:
        return (
            f"this operator ({self.op_name}) requires a paid API key; "
            f"execution is blocked in the open-core ocean-cli wheel.\n"
            f"hint: see https://latentocean.com/protocols for an API key, "
            f"or call the proprietary backend at https://api.latentocean.com/run"
        )


class _PremiumStub:
    def __init__(self, name: str) -> None:
        self.name = name
        self.tier = "premium"

    def run(self, *args, **kwargs):
        raise PremiumOperatorError(self.name)


# Names mirror backend.handbook_runner.premium_gate.OPERATOR_REGISTRY.
PREMIUM_OPS = (

    "reduce.btut",
```

```python
"""Preset name resolution: <namespace>.<preset> → stdlib file + define name.

Lookup order:
    1. Directories in OCEAN_PATH env var (colon-separated on POSIX, semicolon on Windows)
    2. <package>/stdlib/<namespace>.ocean (bundled)
"""
from __future__ import annotations

import argparse
import os
import re
import sys
import tempfile
from pathlib import Path

from ocean_cli.paths import stdlib_dir


class PresetResolutionError(ValueError):
    """Raised when a preset name is malformed or unresolvable."""


_NAME_RE = re.compile(r"^([a-z_][a-z0-9_]*)$")
_DEFINE_RE = re.compile(
    r"^\s*define\s+([a-z_][a-z0-9_]*)\s*\(",
    re.MULTILINE,
)


def parse_preset_name(name: str) -> tuple[str, str]:
    """Split 'namespace.preset' into (namespace, preset). Raise on malformed."""
    if "." not in name:
        raise PresetResolutionError(
            f"preset name '{name}' must contain a dot (e.g., 'pulse.uspto')"
        )
    parts = name.split(".")
    if len(parts) != 2:
        raise PresetResolutionError(
            f"preset name '{name}' must contain exactly one dot"
        )
    ns, preset = parts
    if not _NAME_RE.match(ns):
        raise PresetResolutionError(f"invalid namespace '{ns}'")
    if not _NAME_RE.match(preset):
        raise PresetResolutionError(f"invalid preset name '{preset}'")
    return ns, preset


def _path_dirs() -> list[Path]:
    """Return OCEAN_PATH directories in lookup order."""
    raw = os.environ.get("OCEAN_PATH", "")
    if not raw:
        return []
    sep = ";" if os.name == "nt" else ":"
    return [Path(p) for p in raw.split(sep) if p]


def resolve_stdlib_path(namespace: str) -> Path | None:
    """Find the .ocean file backing a namespace. Returns None if not found."""
    filename = f"{namespace}.ocean"
    for d in _path_dirs():
        candidate = d / filename
        if candidate.is_file():
            return candidate
    bundled = stdlib_dir() / filename
    if bundled.is_file():
        return bundled
    return None


def list_presets(namespace: str) -> list[str]:
    """Return the names of all defines in the given namespace's stdlib file."""
    path = resolve_stdlib_path(namespace)
    if path is None:
        return []
    text = path.read_text(encoding="utf-8")
    return [m.group(1) for m in _DEFINE_RE.finditer(text)]


def list_all_namespaces() -> list[str]:
    """Enumerate every .ocean file in the bundled stdlib (and OCEAN_PATH dirs)."""
    seen: set[str] = set()
    for d in [*_path_dirs(), stdlib_dir()]:
        if not d.is_dir():
            continue
        for f in d.glob("*.ocean"):
            seen.add(f.stem)
    return sorted(seen)


def run_preset(name: str, args: argparse.Namespace) -> int:
    """Resolve <namespace>.<preset> and execute it via the file-run path.

    Implementation:
        1. Parse name → (namespace, preset).
        2. Resolve stdlib path.
        3. Verify the preset exists in that file.
        4. Generate a temp program: `import "<path>" as ns; ns.<preset>(<overrides>)`.
        5. Set cwd to <package_root> so bundled-relative paths resolve.
        6. Dispatch to the file runner.
    """
    try:
        ns, preset = parse_preset_name(name)
    except PresetResolutionError as e:
        print(f"ocean: error: {e}", file=sys.stderr)
        return 1

    path = resolve_stdlib_path(ns)
    if path is None:
        avail = list_all_namespaces()
        print(
            f"ocean: error: unknown namespace '{ns}'; "
            f"available: {', '.join(avail) or '(none)'}",
            file=sys.stderr,
        )
        return 1

    presets = list_presets(ns)
    if preset not in presets:
        print(
            f"ocean: error: unknown preset '{ns}.{preset}'; "
            f"available in '{ns}': {', '.join(presets) or '(none)'}",
            file=sys.stderr,
        )
        return 1

    # Collect overrides from argparse extras. argparse passed --corpus,
    # --target, etc. via getattr; we read all *non-internal* attributes.
    override_args = _collect_overrides(args)
    call_args = ", ".join(f'{k} = {_quote(v)}' for k, v in override_args.items())

    program = (
        f'import "{path.as_posix()}" as {ns}\n'
        f'{ns}.{preset}({call_args})\n'
    )

    pkg_root = path.parent.parent  # package_root() / stdlib / <file>.ocean → package_root
    with tempfile.NamedTemporaryFile(
        mode="w", suffix=".ocean", delete=False, encoding="utf-8"
    ) as tf:
        tf.write(program)
        tmp_path = Path(tf.name)

    try:
        from ocean_cli.commands.run import _run_file

        cur = Path.cwd()
        os.chdir(pkg_root)
        try:
            return _run_file(tmp_path, args)
        finally:
            os.chdir(cur)
    finally:
        try:
            tmp_path.unlink()
        except OSError:
            pass


# argparse parameter overrides land on `args` as attributes. Only the
# user-provided ones (not internal argparse attrs) are forwarded.
_INTERNAL_ATTRS = {"target", "command", "func", "seed"}


def _collect_overrides(args: argparse.Namespace) -> dict[str, object]:
    return {
        k: v
        for k, v in vars(args).items()
        if k not in _INTERNAL_ATTRS and v is not None
    }


def _quote(v: object) -> str:
    if isinstance(v, str):
        return f'"{v}"'
    return repr(v)
```

```python
def _apply_overrides(args: argparse.Namespace) -> None:
    """Parse `--key value` pairs from args.overrides into args attributes."""
    tokens = list(args.overrides or [])
    while tokens:
        tok = tokens.pop(0)
        if not tok.startswith("--"):
            continue
        if "=" in tok:
            key, val = tok[2:].split("=", 1)
        else:
            key = tok[2:]
            val = tokens.pop(0) if tokens else ""
        # Try to coerce to int or float; fall back to string.
        coerced: object = val
        try:
            coerced = int(val)
        except ValueError:
            try:
                coerced = float(val)
            except ValueError:
                pass
        setattr(args, key.replace("-", "_"), coerced)
    args.overrides = []
```

```python
from __future__ import annotations
```

```python
def add_parser(subparsers: argparse._SubParsersAction) -> None:
    p = subparsers.add_parser("list", help="enumerate operators or stdlib presets")
    sub = p.add_subparsers(dest="list_kind", required=True)

    ops_p = sub.add_parser("ops", help="enumerate operators")
    ops_p.add_argument("--free", action="store_true", help="free-tier only")
    ops_p.add_argument("--premium", action="store_true", help="premium only")
    ops_p.set_defaults(func=run_ops)

    stdlib_p = sub.add_parser("stdlib", help="enumerate stdlib presets")
    stdlib_p.add_argument("--namespace", default=None, help="filter to one namespace")
    stdlib_p.set_defaults(func=run_stdlib)


def run_ops(args: argparse.Namespace) -> int:
    from ocean_cli._vendored.backend.handbook_runner.premium_gate import OPERATOR_REGISTRY

    show_free = not args.premium
    show_premium = not args.free

    for name in sorted(OPERATOR_REGISTRY):
        spec = OPERATOR_REGISTRY[name]
        if spec.tier == "free" and not show_free:
            continue
        if spec.tier == "premium" and not show_premium:
            continue
        print(f"  {name:35s}  {spec.tier:8s}  {spec.signature}")
        print(f"  {'':35s}            {spec.summary}")
    return 0


def run_stdlib(args: argparse.Namespace) -> int:

    namespaces = (
        [args.namespace] if args.namespace else list_all_namespaces()

        presets = list_presets(ns)
        if not presets and args.namespace:
            print(f"  {ns}: (no presets found)")
            continue
        print(f"  {ns}:")
        for p in presets:
```

```python
from ocean_cli.paths import package_root


def add_parser(subparsers: argparse._SubParsersAction) -> None:
    p = subparsers.add_parser("new", help="scaffold a new OCEAN program from a template")
    p.add_argument("name", help="name of the new file (without .ocean extension)")
    p.add_argument(
        "--template", default="basic",
        help="template to use (default: basic)",
    )
    p.set_defaults(func=run)


def run(args: argparse.Namespace) -> int:
    target = Path(f"{args.name}.ocean")
    if target.exists():
        print(f"ocean: error: file already exists: {target}", file=sys.stderr)
        return 1

    template_path = package_root() / "templates" / f"{args.template}.ocean"
    if not template_path.is_file():
        print(
            f"ocean: error: unknown template '{args.template}'; "
            f"available: {', '.join(p.stem for p in (package_root() / 'templates').glob('*.ocean'))}",
            file=sys.stderr,
        )
        return 1

    target.write_text(template_path.read_text(encoding="utf-8"), encoding="utf-8")
    print(f"wrote {target}")
    return 0
```

```python
"""Pre-publish IP guard for the ocean-cli wheel.

Run before `python -m build`. The script:
    1. Re-vendors the compiler + free-tier operators from the source tree.
    2. AST-strips any proprietary class definitions that slipped in.
    3. Emits a TRIMMED.txt manifest of what was removed.

Exits non-zero if any proprietary identifier still exists in the
vendored tree after stripping.

Usage:
    python packages/ocean-cli/scripts/vendor_strip.py
"""
from __future__ import annotations

import ast
import shutil
import sys
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parents[3]
PKG = REPO_ROOT / "packages/ocean-cli"
VENDORED = PKG / "src/ocean_cli/_vendored"
SOURCE_OPERATORS = REPO_ROOT / "scripts/operators"
SOURCE_BACKEND = REPO_ROOT / "backend/handbook_runner"

PROPRIETARY_CLASSES = (
    "ContentFP48Embedder",
    "BTUTReducer",
    "TCDRecursiveLoop",
    "DispersionAlignment",
)


def main() -> int:
    if not VENDORED.exists():
        print(f"error: {VENDORED} does not exist; run Task 2 first", file=sys.stderr)
        return 1

    trimmed: list[str] = []
    for py in VENDORED.rglob("*.py"):
        text = py.read_text(encoding="utf-8")
        new_text = _strip_classes(text, PROPRIETARY_CLASSES, removed=trimmed, path=py)
        if new_text != text:
            py.write_text(new_text, encoding="utf-8")

    manifest = PKG / "src/ocean_cli/_vendored/TRIMMED.txt"
    manifest.write_text("\n".join(sorted(trimmed)) + "\n", encoding="utf-8")
    print(f"vendor_strip: removed {len(trimmed)} class(es) from vendored tree")
    return 0


def _strip_classes(source: str, names: tuple[str, ...], *, removed: list[str], path: Path) -> str:
    try:
        tree = ast.parse(source)
    except SyntaxError:
        return source
    new_body: list[ast.stmt] = []
    for node in tree.body:
        if isinstance(node, ast.ClassDef) and node.name in names:
            removed.append(f"{path.relative_to(VENDORED)}::{node.name}")
            continue
        new_body.append(node)
    tree.body = new_body
    return ast.unparse(tree) + "\n"


if __name__ == "__main__":
    sys.exit(main())
```
